\chapter{MATRIZ DE CONSISTENCIA}
\label{anx:matriz}
\begingroup\small\setstretch{1}
\begin{longtable}{@{}P{4.6cm}P{4.5cm}P{4.2cm}P{4.3cm}P{3.6cm}@{}}
\toprule
\textbf{Problema} & \textbf{Objetivos} & \textbf{Hipótesis} & \textbf{Variables e indicadores} & \textbf{Metodología} \\
\midrule
\endhead
\textbf{General:} ¿De qué manera un SGC basado en la norma ISO~9001, centrado en el control de materiales, permite reducir los costos de ejecución de las losas alivianadas con viguetas pretensadas del Edificio de Posgrado de la UATF?
& \textbf{General:} Diseñar un SGC basado en la norma ISO~9001, centrado en el control de materiales, para reducir los costos de ejecución de dichas losas.
& \textbf{General:} El SGC reduce en al menos 5~\% el costo unitario (Bs/\mdos{}, a precios constantes) respecto de la línea base.
& \textbf{X:} SGC (planificación, control de materiales, control del proceso, mejora). \newline \textbf{Y:} costo de ejecución (costo directo, desperdicio, CNC, costo de la calidad, conformidad del hormigón).
& Aplicada; cuantitativa; explicativa. Cuasi-experimental $O_1$--$X$--$O_2$. \\
\midrule
\textbf{PE1:} ¿Cómo se gestionaron la calidad y el control de materiales en la primera losa?
& \textbf{OE1:} Caracterizar la gestión actual e identificar las no conformidades y sus causas.
& (Descriptivo)
& Porcentaje de cumplimiento ISO~9001; número de no conformidades.
& Lista de verificación, análisis documental, Pareto e Ishikawa. \\
\midrule
\textbf{PE2:} ¿Cuál es el desperdicio de materiales y el costo de no calidad de la primera losa?
& \textbf{OE2:} Cuantificar la línea base del índice de desperdicio y del costo de no calidad.
& (Descriptivo: valor de referencia)
& $IDM_i$; Bs/\mdos{}; \%CNC.
& Fichas de materiales, cómputos y planillas. \\
\midrule
\textbf{PE3:} ¿Qué componentes debe tener el SGC para este proceso constructivo?
& \textbf{OE3:} Establecer el plan de la calidad, el PPI, los procedimientos y los registros.
& ---
& Componentes del SGC.
& ISO~9001, ISO~10005, ACI~318-25 y ASTM. \\
\midrule
\textbf{PE4:} ¿En qué medida el SGC reduce el costo y el desperdicio?
& \textbf{OE4 y OE5:} Evaluar el efecto, comprobar la hipótesis y validar el SGC con expertos.
& \textbf{HE1:} reduce el $IDM$. \newline \textbf{HE2:} reduce el CNC.
& $IDM_i$, CNC y Bs/\mdos{}.
& Prueba \emph{t} o Wilcoxon ($\alpha = 0{,}05$); V de Aiken. \\
\bottomrule
\end{longtable}
\endgroup
