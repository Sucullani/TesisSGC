% =====================================================================
%  PREÁMBULO DE LA TESIS: paquetes y formato
%  Formato según la Guía de Normas APA 7.ª edición del repositorio
%  (04_Bibliografia/02_Metodologia_y_Redaccion/Guia_Normas_APA_7ma_Edicion.pdf):
%    - papel carta; márgenes de 2,54 cm en los cuatro lados (pp. 26-27);
%    - Times New Roman de 12 pt (p. 26);
%    - interlineado doble, sin espacio adicional entre párrafos (p. 30);
%    - alineación a la izquierda, sin justificar ni dividir palabras (pp. 31-32);
%    - sangría de 1,27 cm en la primera línea de cada párrafo (p. 32);
%    - número de página en la esquina superior derecha (pp. 27-28);
%    - títulos en negrita: nivel 1 centrado, nivel 2 a la izquierda,
%      nivel 3 a la izquierda en cursiva (pp. 28-29);
%    - tablas y figuras: número en negrita y título en cursiva en la línea
%      siguiente, nota debajo (pp. 17-23);
%    - referencias con sangría francesa de 1,27 cm (p. 33).
%  Los capítulos y secciones se numeran porque así lo exige la estructura
%  del documento de graduación de la Carrera (Taller 1, CIV 400).
% =====================================================================

% --- Títulos de capítulos y secciones (niveles APA) ---------------------
\usepackage{titlesec}
% Nivel 1: centrado, negrita (capítulos)
\titleformat{\chapter}[display]
  {\normalfont\normalsize\bfseries\centering}
  {\MakeUppercase{\chaptertitlename}\ \thechapter}{0pt}{\MakeUppercase}
\titleformat{name=\chapter,numberless}[block]
  {\normalfont\normalsize\bfseries\centering}{}{0pt}{\MakeUppercase}
\titlespacing*{\chapter}{0pt}{0pt}{12pt}
% Nivel 2: a la izquierda, negrita
\titleformat{\section}{\normalfont\normalsize\bfseries\raggedright}{\thesection.}{0.6em}{}
% Nivel 3: a la izquierda, negrita y cursiva
\titleformat{\subsection}{\normalfont\normalsize\bfseries\itshape\raggedright}{\thesubsection.}{0.6em}{}
% Nivel 4: con sangría, negrita, punto final y texto en la misma línea
\titleformat{\subsubsection}[runin]{\normalfont\normalsize\bfseries}{\thesubsubsection.}{0.5em}{}[.]
\titlespacing*{\section}{0pt}{6pt}{0pt}
\titlespacing*{\subsection}{0pt}{6pt}{0pt}
\titlespacing*{\subsubsection}{1.27cm}{3pt}{0.5em}
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{2}
